# fancy-dual-boot theme — text only, no images, nothing third-party.
#
# Install as <esp>/EFI/<dir>/theme/theme.txt and point grub.cfg at it from
# inside a loadfont guard, so a missing font degrades to the plain text menu
# instead of an error:
#
#   if loadfont $prefix/fonts/unicode.pf2; then
#       set gfxmode=auto
#       terminal_output gfxterm
#       set theme=$prefix/theme/theme.txt
#       export theme
#   fi
#
# Icons are optional. GRUB looks for theme/icons/<class>.png, 96x96, where
# <class> comes from --class on the menuentry. Missing icons are cosmetic.
